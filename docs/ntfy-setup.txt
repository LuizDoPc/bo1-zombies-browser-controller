# Self-hosted signaling for bo1z co-op (instead of the public ntfy.sh, which rate-limits per IP address: HTTP 429).
# ntfy only carries the two connection-setup messages per player; game traffic is peer to peer.

# 1. Install ntfy (Debian/Ubuntu; other options: https://docs.ntfy.sh/install/)
sudo mkdir -p /etc/apt/keyrings
curl -fsSL https://archive.heckel.io/apt/pubkey.txt | sudo gpg --dearmor -o /etc/apt/keyrings/archive.heckel.io.gpg
echo "deb [arch=amd64 signed-by=/etc/apt/keyrings/archive.heckel.io.gpg] https://archive.heckel.io/apt debian main" | sudo tee /etc/apt/sources.list.d/archive.heckel.io.list
sudo apt update && sudo apt install ntfy

# 2. /etc/ntfy/server.yml
base-url: "https://your.domain"
listen-http: "127.0.0.1:2586"
behind-proxy: true

sudo systemctl enable --now ntfy

# 3. nginx, in the same server { } block as /bo1z/:
        location /ntfy/ {
                proxy_pass http://127.0.0.1:2586/;
                proxy_http_version 1.1;
                proxy_buffering off;
                proxy_read_timeout 3m;
                proxy_set_header Host $host;
                proxy_set_header X-Forwarded-For $remote_addr;
                add_header Cross-Origin-Resource-Policy same-origin always;
        }
sudo nginx -t && sudo nginx -s reload

# 4. Check: this should print a JSON line with "message":"hi"
curl -s -d hi https://your.domain/ntfy/bo1z-check

# 5. Use it: add &ntfy=https://your.domain/ntfy to the host URL (the share link carries it to joiners),
#    or make it the default in mp/net-config.json:  "ntfy": "https://your.domain/ntfy"  (or edit net-config.json on the server).
